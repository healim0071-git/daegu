function isHealimSuperAdmin() {
  try {
    var raw = localStorage.getItem('healim_auth_user');
    if (!raw) return false;
    var u = JSON.parse(raw);
    return (u.uid === 'healim0071' || u.isAdmin === true || u.grade === 'superadmin' || u.role === 'admin');
  } catch(e) { return false; }

function getDiseaseDetailPageUrl(disease) {
  if (!disease) return '/columns/';
  var d = disease.trim();
  if (d.includes('틱')) return '/tic-disorder/';
  if (d.includes('공황')) return '/panic-disorder/';
  if (d.includes('어지')) return '/dizziness/';
  if (d.includes('자율신경')) return '/dysautonomia/';
  if (d.includes('불면')) return '/insomnia/';
  if (d.includes('ADHD') || d.includes('adhd')) return '/adhd/';
  if (d.includes('불안')) return '/anxiety-disorder/';
  if (d.includes('강박')) return '/ocd/';
  if (d.includes('대인') || d.includes('사회')) return '/social-phobia/';
  if (d.includes('다한')) return '/hyperhidrosis/';
  if (d.includes('갱년기')) return '/climacteric/';
  return '/columns/';
}

var HEALIM_CANONICAL_FAQ_VERSION = '20261030_v105_canonical_sync';
function getRealtimeBoardFaqData() {
  try {
    var storedVer = localStorage.getItem('healim_canonical_faq_version');
    if (storedVer !== HEALIM_CANONICAL_FAQ_VERSION) {
      window.__healimAllowDataReset = true;
      localStorage.removeItem('healim_board_faqs');
      localStorage.removeItem('healim_board_faqs_permanent_backup');
      if (typeof sessionStorage !== 'undefined') sessionStorage.removeItem('healim_board_faqs_backup');
      localStorage.setItem('healim_canonical_faq_version', HEALIM_CANONICAL_FAQ_VERSION);
      window.__healimAllowDataReset = false;
    }
  } catch(e) {}

  var list = defaultFaqData.slice();
  try {
    var savedRaw = localStorage.getItem('healim_board_faqs');
    if (savedRaw) {
      var saved = JSON.parse(savedRaw);
      if (Array.isArray(saved) && saved.length > 0) {
        saved.forEach(function(sItem) {
          if (!sItem || !sItem.id) return;
          var exists = list.some(function(it) { return String(it.id) === String(sItem.id) || (sItem.q && it.q === sItem.q); });
          if (!exists) list.unshift(sItem);
        });
      }
    }
  } catch(e) {}
  return list;
}

function toggleFaqAccordion(id) {
  var ans = document.getElementById('faq-answer-' + id);
  var icon = document.getElementById('faq-icon-' + id);
  if (!ans) return;
  var isOpen = ans.style.display === 'block';
  ans.style.display = isOpen ? 'none' : 'block';
  if (icon) icon.style.transform = isOpen ? 'rotate(0deg)' : 'rotate(180deg)';
}

  if (autoBar) autoBar.style.display = isSuper ? 'flex' : 'none';

  var delBtns = document.querySelectorAll('.faq-admin-del-btn');
  delBtns.forEach(function(b) { b.style.display = isSuper ? 'inline-block' : 'none'; });

  var container = document.getElementById('faqAccordionContainer');
  if (!container) return;

  var list = getRealtimeBoardFaqData();
  var filtered = list.filter(function(item) {
    if (!item) return false;
    var matchCat = (currentFaqCategory === 'ALL' || item.disease === currentFaqCategory);
    var matchKey = (!currentFaqKeyword || (item.q && item.q.includes(currentFaqKeyword)) || (item.a && item.a.includes(currentFaqKeyword)));
    return matchCat && matchKey;
  });

  container.innerHTML = '';
  if (filtered.length === 0) {
    container.innerHTML = '<div style="text-align: center; padding: 40px; color: #94A3B8;">검색된 질문이 없습니다.</div>';
    return;
  }

  filtered.forEach(function(item, idx) {
    var div = document.createElement('div');
    div.className = 'faq-item';
    div.id = 'faq-item-' + item.id;
    div.style.cssText = 'border: 1px solid #E2E8F0; border-radius: 12px; margin-bottom: 14px; background: #FFFFFF; overflow: hidden; box-shadow: 0 2px 8px rgba(0,0,0,0.02); transition: all 0.2s;';

    var linkTxt = item.linkText || ((item.disease || '질환') + ' 정밀클리닉 바로가기');

    var adminDelBtn = isSuper ? '<button type="button" class="faq-admin-del-btn" onclick="event.stopPropagation(); handleAdminDeleteFaq(\'' + item.id + '\');" style="padding: 4px 8px; font-size: 11px; background: #FEE2E2; color: #DC2626; border: 1px solid #FCA5A5; border-radius: 4px; cursor: pointer; font-weight: 700;">삭제</button>' : '';

    div.innerHTML = 
      '<div class="faq-question-bar" onclick="toggleFaqAccordion(\'' + item.id + '\')" style="padding: 18px 20px; display: flex; justify-content: space-between; align-items: center; cursor: pointer; background: #FFFFFF; gap: 14px;">' +
      '  <div style="display: flex; align-items: flex-start; gap: 12px; min-width: 0; flex: 1;">' +
      '    <span style="display: inline-flex; align-items: center; justify-content: center; width: 30px; height: 30px; border-radius: 8px; background: #E6F6F8; color: #008194; font-weight: 800; font-size: 15px; flex-shrink: 0;">Q</span>' +
      '    <div>' +
      '      <div style="font-size: 12px; font-weight: 700; color: #008194; margin-bottom: 4px;">[' + (item.disease || '안내') + ']</div>' +
      '      <h3 style="font-size: 15.5px; font-weight: 700; color: #0F172A; margin: 0; line-height: 1.45; word-break: keep-all;">' + (item.q || '') + '</h3>' +
      '    </div>' +
      '  </div>' +
      '  <div style="display: flex; align-items: center; gap: 8px; flex-shrink: 0;">' +
      '    ' + adminDelBtn +
      '    <span id="faq-icon-' + item.id + '" style="font-size: 18px; color: #64748B; transition: transform 0.2s;">▾</span>' +
      '  </div>' +
      '</div>' +
      '<div id="faq-answer-' + item.id + '" class="faq-answer-pane" style="display: none; padding: 20px 22px; background: #F8FAFB; border-top: 1px solid #F1F5F9; font-size: 14.5px; line-height: 1.75; color: #334155; word-break: keep-all;">' +
      '  <div style="margin-bottom: 16px;">' + (item.a || '').replace(/\n/g, '<br>') + '</div>' +
      '  <div style="display: flex; gap: 10px; flex-wrap: wrap; padding-top: 14px; border-top: 1px dashed #E2E8F0;">' +
      '    <a href="' + linkUrl + '" style="display: inline-flex; align-items: center; gap: 4px; padding: 7px 14px; background: #008194; color: #FFF; font-size: 12.5px; font-weight: 700; border-radius: 6px; text-decoration: none;">' +
      '      <span>' + linkTxt + ' &gt;</span>' +
      '    </a>' +
      '    <a href="https://www.healim.com" target="_blank" rel="noopener noreferrer" style="display: inline-flex; align-items: center; gap: 4px; padding: 7px 14px; background: #FFF; color: #0C4349; border: 1px solid #CBD5E1; font-size: 12.5px; font-weight: 700; border-radius: 6px; text-decoration: none;">' +
      '      <span>전국 16개 지점 안내 &gt;</span>' +
      '    </a>' +
      '  </div>' +
      '</div>';
    container.appendChild(div);
  });
}

function filterFaq() {
  var sel = document.getElementById('faqCategoryFilter');
  var inp = document.getElementById('faqSearchInput');
  if (sel) currentFaqCategory = sel.value;
  if (inp) currentFaqKeyword = inp.value.trim();
  renderFaqList();
}

function openFaqModal() {
  var modal = document.getElementById('faqWriteModal');
  if (modal) modal.style.display = 'flex';
}

function closeFaqModal() {
  var modal = document.getElementById('faqWriteModal');
  if (modal) modal.style.display = 'none';
}

function handleSaveFaq(e) {
  if (e && e.preventDefault) e.preventDefault();
  var disease = document.getElementById('writeFaqDisease').value;
  var q = document.getElementById('writeFaqQuestion').value.trim();
  var a = document.getElementById('writeFaqAnswer').value.trim();

  if (!q || !a) { alert('질문과 답변을 입력해주세요.'); return; }

  var newFaq = {
    id: 'faq-custom-' + Date.now(),
    disease: disease,
    date: new Date().toISOString().slice(0, 10).replace(/-/g, '.'),
    q: q,
    a: a,
    link: getDiseaseDetailPageUrl(disease),
    linkText: disease + ' 정밀클리닉 바로가기'
  };

  var saved = [];
  try {
    var raw = localStorage.getItem('healim_board_faqs');
    if (raw) saved = JSON.parse(raw);
    if (!Array.isArray(saved)) saved = [];
  } catch(e) {}

  saved.unshift(newFaq);
  localStorage.setItem('healim_board_faqs', JSON.stringify(saved));
  closeFaqModal();
  renderFaqList();
  alert('FAQ가 저장되었습니다.');
}

function handleAdminDeleteFaq(id) {
  if (!confirm('정말 이 질문을 삭제하시겠습니까?')) return;
  try {
    var raw = localStorage.getItem('healim_board_faqs');
    var list = raw ? JSON.parse(raw) : [];
    list = list.filter(function(it) { return String(it.id) !== String(id); });
    localStorage.setItem('healim_board_faqs', JSON.stringify(list));

    var delRaw = localStorage.getItem('healim_deleted_faq_ids');
    var delList = delRaw ? JSON.parse(delRaw) : [];
    delList.push(id);
    localStorage.setItem('healim_deleted_faq_ids', JSON.stringify(delList));
  } catch(e) {}
  renderFaqList();
}

function triggerAutoFaqPublishManual() {
  if (typeof window.triggerAutoFaqPublish === 'function') {
    window.triggerAutoFaqPublish();
    setTimeout(renderFaqList, 300);
  } else {
    alert('FAQ 자동 발행 엔진이 준비 중입니다.');
  }
}
