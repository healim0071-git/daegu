  var bShorts = document.getElementById('ytFilterShorts');

  [bAll, bLong, bShorts].forEach(function(b) {
    if (!b) return;
    b.style.background = '#FFF';
    b.style.color = '#475569';
    b.style.borderColor = '#CBD5E1';
  });

  if (type === 'ALL' && bAll) { bAll.style.background = '#146369'; bAll.style.color = '#FFF'; bAll.style.borderColor = '#146369'; }
  if (type === 'long' && bLong) { bLong.style.background = '#146369'; bLong.style.color = '#FFF'; bLong.style.borderColor = '#146369'; }
  if (type === 'shorts' && bShorts) { bShorts.style.background = '#146369'; bShorts.style.color = '#FFF'; bShorts.style.borderColor = '#146369'; }

  var cards = document.querySelectorAll('.youtube-card');
  cards.forEach(function(card) {
    var cType = card.getAttribute('data-ytype');
    if (type === 'ALL' || cType === type) {
      card.style.display = 'flex';
    } else {
      card.style.display = 'none';
    }
  });
}

function openYoutubeModal(vid, title) {
  var modal = document.getElementById('youtubePlayerModal');
  var titleEl = document.getElementById('youtubeModalTitle');
  var container = document.getElementById('youtubeModalIframeContainer');
  var linkEl = document.getElementById('youtubeModalDirectLink');

  if (titleEl) titleEl.textContent = title || '해아림TV 영상';
  if (linkEl) linkEl.href = 'https://www.youtube.com/watch?v=' + encodeURIComponent(vid);

  if (container) {
    container.innerHTML = '<iframe style="position: absolute; inset: 0; width: 100%; height: 100%; border: none;" src="https://www.youtube.com/embed/' + encodeURIComponent(vid) + '?autoplay=1&rel=0" allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture" allowfullscreen></iframe>';
  }

  if (modal) {
    modal.style.display = 'flex';
    document.body.style.overflow = 'hidden';
  }
}

function closeYoutubeModal() {
  var modal = document.getElementById('youtubePlayerModal');
  var container = document.getElementById('youtubeModalIframeContainer');
  if (container) container.innerHTML = '';
  if (modal) {
    modal.style.display = 'none';
    document.body.style.overflow = '';
  }
}

document.addEventListener('keydown', function(e) {
  if (e.key === 'Escape') closeYoutubeModal();
});

document.addEventListener('DOMContentLoaded', function() {
  var params = new URLSearchParams(window.location.search);
  var vId = params.get('vid') || params.get('id');
  if (vId) {
    var found = defaultYoutubeData.find(function(y) { return y.vid === vId || String(y.id) === vId; });
    setTimeout(function() {
      openYoutubeModal(vId, found ? found.title : '해아림TV 영상');
    }, 150);
  }
});
